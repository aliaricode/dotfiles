#   spotifyd                  Spotify player device ("quickshell-player") for MusicLibraryPopup
#   libqalculate              launcher "=" calculator (qalc)
#   unicode-emoji             launcher ":" emoji picker (emoji-test.txt)
#   fd                        launcher "/" file search

#   slurp                     region selection for recordings
